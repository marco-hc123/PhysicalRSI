
import json as _json
import time as _time

def _publish(message):
    with open('/output/result.json', 'w') as stream:
        _json.dump(message, stream, allow_nan=False)

class _Robot:
    def __init__(self):
        self._index = 0

    def __getattr__(self, name):
        def call(*args, **kwargs):
            index = self._index
            self._index += 1
            _publish(dict(kind='call', id=index, method=name,
                          args=args, kwargs=kwargs))
            while True:
                with open('/response.json') as stream:
                    reply = _json.load(stream)
                if reply.get('id') == index:
                    value = reply['value']
                    # Only a trusted, completed pre-action planner rejection is
                    # recoverable. Transport/partial-action failures still abort.
                    if isinstance(value, dict) and value.get('primitive_error') == 'no_collision_free_path':
                        raise RuntimeError('No collision-free path; no action executed by this call')
                    return value
                _time.sleep(0.01)
        return call

_input = _json.load(open('/input.json'))
_scope = {}
exec(compile(_input['source'], '<candidate>', 'exec'), _scope)
_value = _scope['policy'](_Robot(), _input['memory'])
_publish(dict(kind='result', value=_value, memory=_input['memory']))
